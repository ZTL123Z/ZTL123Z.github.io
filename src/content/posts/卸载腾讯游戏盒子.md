# 使用管理员运行代码删除（成功）（不建议尝试，电脑会被删除文件）

```powershell
# 1. 强制关闭 GameBox 相关进程
Stop-Process -Name "GameBox" -Force -ErrorAction SilentlyContinue
Stop-Process -Name "GameBox.exe" -Force -ErrorAction SilentlyContinue
Stop-Process -Name "Utility" -Force -ErrorAction SilentlyContinue

# 2. 跳过无法使用的 Defender 命令（你系统不支持）

# 3. 安全删除残留目录（不存在也不会报错）
if (Test-Path "C:\ProgramData\client") {
    takeown /f "C:\ProgramData\client" /r /d y
    icacls "C:\ProgramData\client" /grant administrators:F /t
    Remove-Item -Path "C:\ProgramData\client" -Recurse -Force
}

# 4. 清理常见残留位置（自动扫描删除）
$paths = @(
    "C:\ProgramData\GameBox",
    "C:\Program Files\GameBox",
    "C:\Program Files (x86)\GameBox",
    $env:APPDATA + "\GameBox",
    $env:LOCALAPPDATA + "\GameBox"
)

foreach ($path in $paths) {
    if (Test-Path $path) {
        Write-Host "正在清理：$path"
        takeown /f "$path" /r /d y
        icacls "$path" /grant administrators:F /t
        Remove-Item -Path "$path" -Recurse -Force
    }
}

Write-Host "`n✅ 清理完成！GameBox 相关残留已全部删除"
```
